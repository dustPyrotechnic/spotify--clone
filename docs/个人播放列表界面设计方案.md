# 个人播放列表界面设计方案 (MVC 架构)

> 本文档定义了个人播放列表界面的整体设计方案，采用 MVC 架构模式
> 版本: 1.1
> 日期: 2026-02-20

---

## 1. 设计目标

打造一个符合 Spotify 风格的个人播放列表管理界面，支持：
- 展示用户所有的播放列表
- 搜索过滤播放列表
- 创建、筛选、切换展示方式等功能
- 点击播放列表进入详细页面（复用现有详细页面）

---

## 6. 核心功能设计

### 6.1 搜索功能 (Controller → Model → View)

```objc
#pragma mark - Controller: UISearchResultsUpdating
@interface XCPersonalViewController () <UISearchResultsUpdating>
@end

@implementation XCPersonalViewController
- (void)updateSearchResultsForSearchController:(UISearchController*)searchController {
    NSString* searchText = searchController.searchBar.text;
    // Controller 调用 Model 进行过滤
    [self.model filterPlaylistsWithSearchText:searchText];
    // Controller 通知 View 刷新
    [self.mainView.collectionView reloadData];
}
@end

#pragma mark - Model: 过滤逻辑
@implementation XCPersonalModel
- (void)filterPlaylistsWithSearchText:(NSString*)searchText {
    if (searchText.length == 0) {
        _filteredPlaylists = [_allPlaylists mutableCopy];
    } else {
        NSPredicate* predicate = [NSPredicate predicateWithFormat:@"name CONTAINS[cd] %@", searchText];
        _filteredPlaylists = [[_allPlaylists filteredArrayUsingPredicate:predicate] mutableCopy];
    }
    // 保持当前排序
    [self applyCurrentSort];
}
@end
```

### 6.2 筛选功能 (Controller → Model → View)

### 6.3 视图切换 (Controller 协调)

```objc
// Controller 层处理视图切换
- (void)toggleViewMode:(id)sender {
    XCPlaylistViewMode newMode = (self.currentViewMode == XCPlaylistViewModeGrid) ? 
                                 XCPlaylistViewModeList : XCPlaylistViewModeGrid;
    
    self.currentViewMode = newMode;
    
    // 更新按钮图标
    NSString* iconName = (newMode == XCPlaylistViewModeGrid) ? @"list.bullet" : @"square.grid.2x2";
    self.viewModeButton.image = [UIImage systemImageNamed:iconName];
    
    // 切换 CollectionView Layout
    UICollectionViewLayout* newLayout = (newMode == XCPlaylistViewModeGrid) ? 
        [self createGridLayout] : [self createListLayout];
    
    [self.mainView.collectionView setCollectionViewLayout:newLayout animated:YES];
    
    // 保存用户偏好
    [[NSUserDefaults standardUserDefaults] setInteger:newMode forKey:@"PlaylistViewMode"];
}
```

## 7. 交互设计

### 7.1 点击播放列表 (Controller 处理跳转)

```objc
#pragma mark - Controller: UICollectionViewDelegate
- (void)collectionView:(UICollectionView*)collectionView didSelectItemAtIndexPath:(NSIndexPath*)indexPath {
    // 从 Model 获取数据
    XC_YYAlbumData* playlist = self.model.filteredPlaylists[indexPath.row];
    
    // Controller 创建并配置详细页面
    XCALbumDetailViewController* detailVC = [[XCALbumDetailViewController alloc] init];
    detailVC.model.mainImaUrl = playlist.coverImgUrl;
    detailVC.model.playerlistName = playlist.name;
    
    // Controller 发起网络请求获取详情
    [[XCNetworkManager sharedInstance] getDetailOfAlbumFromWY:detailVC.model.playerList 
                                                    ofAlbumId:playlist.albumId 
                                               withCompletion:^(BOOL success) {
        if (success) {
            // Controller 执行页面跳转
            [self.navigationController pushViewController:detailVC animated:YES];
        }
    }];
}
```

### 7.2 长按菜单 (Controller 处理)

```objc
- (void)handleLongPress:(UILongPressGestureRecognizer*)gesture {
    if (gesture.state != UIGestureRecognizerStateBegan) return;
    
    CGPoint point = [gesture locationInView:self.mainView.collectionView];
    NSIndexPath* indexPath = [self.mainView.collectionView indexPathForItemAtPoint:point];
    if (!indexPath) return;
    
    XC_YYAlbumData* playlist = self.model.filteredPlaylists[indexPath.row];
    
    // Controller 显示操作菜单
    UIAlertController* alert = [UIAlertController alertControllerWithTitle:playlist.name
                                                                   message:nil
                                                            preferredStyle:UIAlertControllerStyleActionSheet];
    
    [alert addAction:[UIAlertAction actionWithTitle:@"编辑详情" style:UIAlertActionStyleDefault handler:^(UIAlertAction * _Nonnull action) {
        [self editPlaylist:playlist];
    }]];
    
    [alert addAction:[UIAlertAction actionWithTitle:@"删除" style:UIAlertActionStyleDestructive handler:^(UIAlertAction * _Nonnull action) {
        [self.model deletePlaylistWithId:playlist.albumId completion:^(BOOL success) {
            if (success) {
                [self.mainView.collectionView reloadData];
            }
        }];
    }]];
    
    [self presentViewController:alert animated:YES completion:nil];
}
```

#import "XCPersonalTableViewCell.h"
#import "XCALbumDetailViewController.h"

@interface XCPersonalViewController () <UICollectionViewDataSource, 
                                        UICollectionViewDelegate,
                                        UISearchResultsUpdating>
@property (nonatomic, strong) XCPersonalView* mainView;
@property (nonatomic, strong) XCPersonalModel* model;
@property (nonatomic, strong) UISearchController* searchController;
@property (nonatomic, assign) XCPlaylistViewMode currentViewMode;
@end

@implementation XCPersonalViewController

#pragma mark - Life Cycle
- (void)viewDidLoad {
    [super viewDidLoad];
    
    // 1. 初始化 Model
    self.model = [XCPersonalModel sharedInstance];
    
    // 2. 初始化 View
    [self setupView];
    
    // 3. 设置导航栏
    [self setupNavigationBar];
    
    // 4. 加载数据
    [self loadData];
}

#pragma mark - Setup
- (void)setupView {
    self.mainView = [[XCPersonalView alloc] init];
    self.mainView.collectionView.dataSource = self;
    self.mainView.collectionView.delegate = self;
    [self.view addSubview:self.mainView];
    
    [self.mainView mas_makeConstraints:^(MASConstraintMaker *make) {
        make.edges.equalTo(self.view);
    }];
}

- (void)setupNavigationBar {
    // 创建按钮
    UIBarButtonItem* createBtn = [[UIBarButtonItem alloc] 
        initWithImage:[UIImage systemImageNamed:@"plus.circle.fill"]
                style:UIBarButtonItemStylePlain
               target:self
               action:@selector(createPlaylist:)];
    
    // 筛选按钮
    UIBarButtonItem* filterBtn = [[UIBarButtonItem alloc]
        initWithImage:[UIImage systemImageNamed:@"line.3.horizontal.decrease.circle"]
                style:UIBarButtonItemStylePlain
               target:self
               action:@selector(showSortMenu:)];
    
    // 视图切换按钮
    UIBarButtonItem* viewModeBtn = [[UIBarButtonItem alloc]
        initWithImage:[UIImage systemImageNamed:@"square.grid.2x2"]
                style:UIBarButtonItemStylePlain
               target:self
               action:@selector(toggleViewMode:)];
    
    self.navigationItem.leftBarButtonItems = @[createBtn, filterBtn, viewModeBtn];
    
    // 搜索栏
    self.searchController = [[UISearchController alloc] initWithSearchResultsController:nil];
    self.searchController.searchResultsUpdater = self;
    self.searchController.obscuresBackgroundDuringPresentation = NO;
    self.searchController.searchBar.placeholder = @"搜索播放列表";
    self.navigationItem.searchController = self.searchController;
}

- (void)loadData {
    __weak typeof(self) weakSelf = self;
    [self.model loadPlaylistsFromLocalWithCompletion:^(BOOL success) {
        dispatch_async(dispatch_get_main_queue(), ^{
            [weakSelf.mainView.collectionView reloadData];
        });
    }];
}

#pragma mark - Actions
- (void)createPlaylist:(id)sender {
    UIAlertController* alert = [UIAlertController alertControllerWithTitle:@"新建播放列表"
                                                                   message:nil
                                                            preferredStyle:UIAlertControllerStyleAlert];
    
    [alert addTextFieldWithConfigurationHandler:^(UITextField * _Nonnull textField) {
        textField.placeholder = @"播放列表名称";
    }];
    
    __weak typeof(self) weakSelf = self;
    [alert addAction:[UIAlertAction actionWithTitle:@"创建" style:UIAlertActionStyleDefault handler:^(UIAlertAction * _Nonnull action) {
        NSString* name = alert.textFields.firstObject.text;
        if (name.length > 0) {
            [weakSelf.model createPlaylistWithName:name completion:^(BOOL success, NSString* playlistId) {
                dispatch_async(dispatch_get_main_queue(), ^{
                    [weakSelf.mainView.collectionView reloadData];
                });
            }];
        }
    }]];
    
    [alert addAction:[UIAlertAction actionWithTitle:@"取消" style:UIAlertActionStyleCancel handler:nil]];
    
    [self presentViewController:alert animated:YES completion:nil];
}

- (void)showSortMenu:(id)sender {
    UIAlertController* alert = [UIAlertController alertControllerWithTitle:@"排序方式"
                                                                   message:nil
                                                            preferredStyle:UIAlertControllerStyleActionSheet];
    
    __weak typeof(self) weakSelf = self;
    [alert addAction:[UIAlertAction actionWithTitle:@"最近创建" style:UIAlertActionStyleDefault handler:^(UIAlertAction * _Nonnull action) {
        [weakSelf.model sortPlaylistsByType:XCPlaylistSortTypeCreateDateDesc];
        [weakSelf.mainView.collectionView reloadData];
    }]];
    
    [alert addAction:[UIAlertAction actionWithTitle:@"名称 A-Z" style:UIAlertActionStyleDefault handler:^(UIAlertAction * _Nonnull action) {
        [weakSelf.model sortPlaylistsByType:XCPlaylistSortTypeNameAsc];
        [weakSelf.mainView.collectionView reloadData];
    }]];
    
    [alert addAction:[UIAlertAction actionWithTitle:@"取消" style:UIAlertActionStyleCancel handler:nil]];
    
    [self presentViewController:alert animated:YES completion:nil];
}

- (void)toggleViewMode:(id)sender {
    // 切换视图模式逻辑...
}

#pragma mark - UISearchResultsUpdating
- (void)updateSearchResultsForSearchController:(UISearchController*)searchController {
    NSString* searchText = searchController.searchBar.text;
    [self.model filterPlaylistsWithSearchText:searchText];
    [self.mainView.collectionView reloadData];
}

#pragma mark - UICollectionViewDataSource
- (NSInteger)collectionView:(UICollectionView*)collectionView numberOfItemsInSection:(NSInteger)section {
    return self.model.filteredPlaylists.count;
}

- (UICollectionViewCell*)collectionView:(UICollectionView*)collectionView 
                 cellForItemAtIndexPath:(NSIndexPath*)indexPath {
    
    XC_YYAlbumData* playlist = self.model.filteredPlaylists[indexPath.row];
    
    if (self.currentViewMode == XCPlaylistViewModeGrid) {
        XCPersonalGridCell* cell = [collectionView dequeueReusableCellWithReuseIdentifier:@"GridCell" 
                                                                             forIndexPath:indexPath];
        [cell configureWithPlaylist:playlist];
        return cell;
    } else {
        XCPersonalTableViewCell* cell = [collectionView dequeueReusableCellWithReuseIdentifier:@"ListCell" 
                                                                                  forIndexPath:indexPath];
        [cell configureWithPlaylist:playlist];
        return cell;
    }
}

#pragma mark - UICollectionViewDelegate
- (void)collectionView:(UICollectionView*)collectionView didSelectItemAtIndexPath:(NSIndexPath*)indexPath {
    XC_YYAlbumData* playlist = self.model.filteredPlaylists[indexPath.row];
    
    // Controller 负责页面跳转
    XCALbumDetailViewController* detailVC = [[XCALbumDetailViewController alloc] init];
    detailVC.model.mainImaUrl = playlist.coverImgUrl;
    detailVC.model.playerlistName = playlist.name;
    
    [[XCNetworkManager sharedInstance] getDetailOfAlbumFromWY:detailVC.model.playerList 
                                                    ofAlbumId:playlist.albumId 
                                               withCompletion:^(BOOL success) {
        if (success) {
            [self.navigationController pushViewController:detailVC animated:YES];
        }
    }];
}

@end
