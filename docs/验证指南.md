## 方法二：创建验证页面

创建一个专门的验证页面，可以直观地查看结果：

### 1. 创建验证按钮

在 `HomePageViewController.m` 中添加测试按钮：

```objc
#ifdef DEBUG
#import "8. 个人播放列表界面/Model/XCPersonalModelTestRunner.h"
#import "8. 个人播放列表界面/View/XCPersonalViewTest.h"
#endif

@implementation HomePageViewController

- (void)viewDidLoad {
    [super viewDidLoad];
    
    // ... 原有代码 ...
    
#ifdef DEBUG
    [self addTestButtons];
#endif
}

#ifdef DEBUG
- (void)addTestButtons {
    // 在右上角添加测试按钮
    UIBarButtonItem* testButton = [[UIBarButtonItem alloc] 
        initWithTitle:@"测试" 
        style:UIBarButtonItemStylePlain 
        target:self 
        action:@selector(showTestMenu:)];
    testButton.tintColor = [UIColor systemOrangeColor];
    
    if (self.navigationItem.rightBarButtonItems) {
        NSMutableArray* items = [self.navigationItem.rightBarButtonItems mutableCopy];
        [items addObject:testButton];
        self.navigationItem.rightBarButtonItems = items;
    } else {
        self.navigationItem.rightBarButtonItem = testButton;
    }
}

- (void)showTestMenu:(id)sender {
    UIAlertController* alert = [UIAlertController alertControllerWithTitle:@"测试菜单"
                                                                   message:nil
                                                            preferredStyle:UIAlertControllerStyleActionSheet];
    
    // Phase 1 测试
    [alert addAction:[UIAlertAction actionWithTitle:@"运行 Phase 1 测试 (Model)" 
                                              style:UIAlertActionStyleDefault 
                                            handler:^(UIAlertAction * _Nonnull action) {
        [XCPersonalModelTestRunner runTestsAndShowResultInViewController:self];
    }]];
    
    // Phase 2 测试
    [alert addAction:[UIAlertAction actionWithTitle:@"运行 Phase 2 测试 (View)" 
                                              style:UIAlertActionStyleDefault 
                                            handler:^(UIAlertAction * _Nonnull action) {
        [XCPersonalViewTest runAllTests];
        
        // 显示结果
        NSDictionary* stats = [XCPersonalViewTest testStatistics];
        NSString* msg = [NSString stringWithFormat:@"通过: %@/%@", 
                        stats[@"passed"], stats[@"total"]];
        UIAlertController* result = [UIAlertController alertControllerWithTitle:@"Phase 2 测试结果"
                                                                        message:msg
                                                                 preferredStyle:UIAlertControllerStyleAlert];
        [result addAction:[UIAlertAction actionWithTitle:@"确定" style:UIAlertActionStyleDefault handler:nil]];
        [self presentViewController:result animated:YES completion:nil];
    }]];
    
    // 查看个人播放列表界面（实际功能预览）
    [alert addAction:[UIAlertAction actionWithTitle:@"预览播放列表界面" 
                                              style:UIAlertActionStyleDefault 
                                            handler:^(UIAlertAction * _Nonnull action) {
        [self showPlaylistPreview];
    }]];
    
    [alert addAction:[UIAlertAction actionWithTitle:@"取消" style:UIAlertActionStyleCancel handler:nil]];
    
    [self presentViewController:alert animated:YES completion:nil];
}

- (void)showPlaylistPreview {
    // 创建预览控制器
    UIViewController* previewVC = [[UIViewController alloc] init];
    previewVC.title = @"播放列表预览";
    previewVC.view.backgroundColor = [UIColor systemBackgroundColor];
    
    // 添加测试数据
    XCPersonalModel* model = [XCPersonalModel sharedInstance];
    [model addTestDataForTesting:5];
    
    // 创建 View
    XCPersonalView* personalView = [[XCPersonalView alloc] init];
    personalView.frame = previewVC.view.bounds;
    personalView.autoresizingMask = UIViewAutoresizingFlexibleWidth | UIViewAutoresizingFlexibleHeight;
    
    // 注册 Cells
    [personalView.collectionView registerClass:[XCPersonalGridCell class] 
                    forCellWithReuseIdentifier:@"GridCell"];
    [personalView.collectionView registerClass:[XCPersonalListCell class] 
                    forCellWithReuseIdentifier:@"ListCell"];
    
    // 简单数据源（仅用于预览）
    personalView.collectionView.dataSource = self;
    personalView.collectionView.delegate = self;
    
    [previewVC.view addSubview:personalView];
    
    // 添加切换按钮
    UIBarButtonItem* toggleBtn = [[UIBarButtonItem alloc] initWithTitle:@"切换" 
                                                                   style:UIBarButtonItemStylePlain 
                                                                  target:personalView 
                                                                  action:@selector(toggleViewModeAnimated:)];
    previewVC.navigationItem.rightBarButtonItem = toggleBtn;
    
    [self.navigationController pushViewController:previewVC animated:YES];
}

#pragma mark - 预览数据源
- (NSInteger)collectionView:(UICollectionView*)collectionView numberOfItemsInSection:(NSInteger)section {
    return [XCPersonalModel sharedInstance].allPlaylists.count;
}

- (UICollectionViewCell*)collectionView:(UICollectionView*)collectionView 
                 cellForItemAtIndexPath:(NSIndexPath*)indexPath {
    
    XCPersonalModel* model = [XCPersonalModel sharedInstance];
    XC_YYAlbumData* playlist = model.allPlaylists[indexPath.row];
    XCLocalPlaylistInfo* info = [model infoForPlaylist:playlist.albumId];
    
    // 根据当前模式返回不同 Cell
    XCPersonalView* personalView = (XCPersonalView*)collectionView.superview;
    if (personalView.currentViewMode == XCPlaylistViewModeGrid) {
        XCPersonalGridCell* cell = [collectionView dequeueReusableCellWithReuseIdentifier:@"GridCell" 
                                                                             forIndexPath:indexPath];
        [cell configureWithPlaylist:playlist info:info];
        return cell;
    } else {
        XCPersonalListCell* cell = [collectionView dequeueReusableCellWithReuseIdentifier:@"ListCell" 
                                                                             forIndexPath:indexPath];
        [cell configureWithPlaylist:playlist info:info];
        return cell;
    }
}

#endif

@end
```

### Phase 2 (View) 手动验证

```objc
#import "8. 个人播放列表界面/View/XCPersonalView.h"
#import "8. 个人播放列表界面/View/cells/XCPersonalGridCell.h"
#import "8. 个人播放列表界面/View/cells/XCPersonalListCell.h"
#import "8. 个人播放列表界面/View/cells/XCPersonalEmptyView.h"
#import "8. 个人播放列表界面/View/cells/XCPersonalLoginView.h"

- (void)testViewManually {
    // 1. 创建主视图
    XCPersonalView* view = [[XCPersonalView alloc] initWithFrame:self.view.bounds];
    [self.view addSubview:view];
    
    // 2. 注册 Cells
    [view.collectionView registerClass:[XCPersonalGridCell class] 
            forCellWithReuseIdentifier:@"GridCell"];
    [view.collectionView registerClass:[XCPersonalListCell class] 
            forCellWithReuseIdentifier:@"ListCell"];
    
    // 3. 测试视图模式切换
    dispatch_after(dispatch_time(DISPATCH_TIME_NOW, (int64_t)(2 * NSEC_PER_SEC)), dispatch_get_main_queue(), ^{
        NSLog(@"切换到列表模式");
        [view switchToListLayoutAnimated:YES];
        
        dispatch_after(dispatch_time(DISPATCH_TIME_NOW, (int64_t)(2 * NSEC_PER_SEC)), dispatch_get_main_queue(), ^{
            NSLog(@"切换到网格模式");
            [view switchToGridLayoutAnimated:YES];
        });
    });
    
    // 4. 测试空状态
    dispatch_after(dispatch_time(DISPATCH_TIME_NOW, (int64_t)(6 * NSEC_PER_SEC)), dispatch_get_main_queue(), ^{
        NSLog(@"显示空状态");
        [view showEmptyView:YES];
        
        dispatch_after(dispatch_time(DISPATCH_TIME_NOW, (int64_t)(2 * NSEC_PER_SEC)), dispatch_get_main_queue(), ^{
            NSLog(@"隐藏空状态");
            [view showEmptyView:NO];
        });
    });
    
    // 5. 测试登录状态
    dispatch_after(dispatch_time(DISPATCH_TIME_NOW, (int64_t)(10 * NSEC_PER_SEC)), dispatch_get_main_queue(), ^{
        NSLog(@"显示登录视图");
        [view showLoginView:YES];
        
        dispatch_after(dispatch_time(DISPATCH_TIME_NOW, (int64_t)(2 * NSEC_PER_SEC)), dispatch_get_main_queue(), ^{
            NSLog(@"隐藏登录视图");
            [view showLoginView:NO];
        });
    });
}
```

## 常见问题排查

### 问题 1: 编译错误 - 找不到头文件

**解决方案：**
```objc
// 使用完整路径导入
#import "8. 个人播放列表界面/Model/XCPersonalModel.h"
#import "8. 个人播放列表界面/View/XCPersonalView.h"

// 或在项目的 Header Search Paths 中添加:
// $(SRCROOT)/Spotify - clone
```

### 问题 2: 测试运行时崩溃

**可能原因：**
- 数据模型未正确归档/解档
- 检查 `XC_YYAlbumData` 和 `XCLocalPlaylistInfo` 的 `NSSecureCoding` 实现

### 问题 3: 图片加载失败

**解决方案：**
```objc
// 确保导入 SDWebImage
#import <SDWebImage/SDWebImage.h>

// 检查占位图
[cell.coverImageView sd_setImageWithURL:url 
                       placeholderImage:[UIImage systemImageNamed:@"music.note"]];
```

### 问题 4: 布局不正确

**解决方案：**
- 确保 `Masonry` 框架已正确链接
- 检查 `layoutSubviews` 中是否调用了 `updateLayoutItemSize`
